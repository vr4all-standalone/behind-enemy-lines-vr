cmake_minimum_required(VERSION 3.22)
project(bel_vr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_subdirectory(vendor/soundtouch)
add_library(bel_audio SHARED native/audio_stretch.cpp)
target_compile_definitions(bel_audio PRIVATE BEL_AUDIO_BUILD)
target_link_libraries(bel_audio PRIVATE bel_soundtouch Threads::Threads)
set_target_properties(bel_audio PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
 set_target_properties(bel_audio PROPERTIES PREFIX "")
endif()
add_library(bel_xr SHARED native/xr_host.cpp native/gl_scene.cpp native/gl_context.cpp native/presentation.cpp
 native/shared/xr_panel.cpp native/shared/ui_compat.cpp native/shared/glb_mesh.cpp vendor/glad/src/gl.c)
target_include_directories(bel_xr PRIVATE native native/shared vendor vendor/stb vendor/glad/include vendor/glm vendor/openxr/include)
if(WIN32)
 set_target_properties(bel_xr PROPERTIES PREFIX "")
 target_link_libraries(bel_xr PRIVATE opengl32 gdi32 user32)
 target_compile_definitions(bel_xr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
